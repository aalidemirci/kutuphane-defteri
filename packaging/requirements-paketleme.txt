# ---------- Paketleyici ----------

# ---------- Pencere motoru sarmalayıcısı ----------

# ---------- Windows: WebView2 köprüsü ----------
# pywebview'ın edgechromium arka ucu .NET birlikte çalışabilirliği üzerinden

# bağımlılığıdır, six pystray'in uyumluluk katmanıdır (`six.moves`). Linux'ta
# tepsi Qt'dir (QSystemTrayIcon), bu yüzden işaret yalnız win32.
# LİSANS: pystray LGPLv3'tür — lisans metni VE kaynağı pakete girer (F12;
# packaging/README.md "Masaüstü zinciri"). six MIT'tir.
# Masaüstü zinciri: bu satırlar ↔ spec `if WINDOWS:` bloğu ↔
# `giris.py::DESKTOP_RUNTIME_MODULES`; eşitliği test_spec_kapsami.py denetler.
pystray==0.19.5 ; sys_platform == "win32"
six==1.17.0 ; sys_platform == "win32"

# ---------- Linux/Pardus: Qt WebEngine ----------
# KS kararı (K17): WebKitGTK/PyGObject yolu ELENDİ (typelib paketleme + Pardus

PyQt5==5.15.11 ; sys_platform == "linux"
